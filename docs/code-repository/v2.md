## Part 1: The Fortified, Enterprise-Ready Implementation Plan

To be brutally honest: the previous implementation was not production-ready for an enterprise client. While it successfully shifted the infrastructure from a dangerous host-process model to a sandboxed Kubernetes model, it contained critical flaws (409 creation race conditions, 502/503 routing blind spots, missing resource quotas, and command injection vulnerabilities). This hardened implementation introduces an in-memory lock map, a resilient polling loop for pod readiness, strict resource allocations, and secure multi-argument configuration models leveraging `ghcr.io`.

### Phase 2: DNS Sanitization Helper

Kubernetes resource names must comply with RFC 1123 (lowercase alphanumeric, characters, and `-`). RIDs contain dots (`.`) and underscores (`_`) which will trigger a 400 API error.

* **File to Create/Update:** `src/utils/k8sSanitizer.ts`
* **Logic:**
```typescript
export function sanitizeK8sResourceName(rid: string, branch: string): string {
  const cleanRid = rid.toLowerCase().replace(/[^a-z0-9]/g, '-');
  const cleanBranch = branch.toLowerCase().replace(/[^a-z0-9]/g, '-');
  return `ws-${cleanRid}-${cleanBranch}`.substring(0, 63).replace(/-+$/, '');
}
```

### Phase 4: Refactoring `src/services/workspaceBroker.ts`

Completely remove `child_process.spawn`. We will implement a non-blocking configuration lookup loop with concurrency locks, and if the target pod does not exist, manifest it securely.

```typescript
import * as k8s from '@kubernetes/client-node';
import { sanitizeK8sResourceName } from '../utils/k8sSanitizer';

const kc = new k8s.KubeConfig();
kc.loadFromDefault();
const k8sCoreApi = kc.makeApiClient(k8s.CoreV1Api);
const NAMESPACE = process.env.WORKSPACE_NAMESPACE || 'telos-workspaces';
const GHCR_ORG = process.env.GHCR_ORG || 'your-github-org-or-username';

// CRITICAL: In-memory lock map prevents multi-request race conditions (409 AlreadyExists)
const activeCreationLocks = new Map<string, Promise<string>>();

export async function getOrCreateWorkspaceSession(rid: string, branch: string): Promise<string> {
  const resourceName = sanitizeK8sResourceName(rid, branch);
  const lockKey = `${rid}:${branch}`;

  // If a creation process is already running for this workspace, append to that execution chain
  if (activeCreationLocks.has(lockKey)) {
    return activeCreationLocks.get(lockKey)!;
  }

  const creationPromise = (async () => {
    try {
      // 1. Verify if the target pod exists and is actively running
      const podStatus = await k8sCoreApi.readNamespacedPod(resourceName, NAMESPACE);
      const isRunning = podStatus.body.status?.phase === 'Running';
      
      if (isRunning) {
        return `http://${resourceName}.${NAMESPACE}.svc.cluster.local:8080`;
      }
      
      // If pod exists but is trapped in an unhealthy state, delete it to trigger self-healing
      if (podStatus.body.status?.phase === 'Failed' || podStatus.body.status?.phase === 'Unknown') {
        await k8sCoreApi.deleteNamespacedPod(resourceName, NAMESPACE);
      }
    } catch (error: any) {
      if (error.statusCode !== 404) {
        throw new Error(`Kubernetes cluster communication failure: ${error.message}`);
      }
    }

    // 2. Pod does not exist. Orchestrate production infrastructure safely.
    await provisionHardenedPodAndService(resourceName, rid, branch);

    // 3. Block routing return until the endpoints are active (Prevents 502/503 errors)
    await waitForPodReadiness(resourceName, 30); // 30-second strict timeout policy

    return `http://${resourceName}.${NAMESPACE}.svc.cluster.local:8080`;
  })();

  // Register the lock, execute the pipeline, and clear the lock when finished
  activeCreationLocks.set(lockKey, creationPromise);
  try {
    return await creationPromise;
  } finally {
    activeCreationLocks.delete(lockKey);
  }
}

async function waitForPodReadiness(resourceName: string, maxRetries: number): Promise<void> {
  for (let i = 0; i < maxRetries; i++) {
    try {
      const statusRes = await k8sCoreApi.readNamespacedPodStatus(resourceName, NAMESPACE);
      const phase = statusRes.body.status?.phase;
      const conditions = statusRes.body.status?.conditions || [];
      const isReady = conditions.some(c => c.type === 'Ready' && c.status === 'True');

      if (phase === 'Running' && isReady) {
        return; // Endpoint is fully active and safe to route user traffic to
      }
    } catch (e) {
      // Ignore intermediate lookup errors during initialization transitions
    }
    await new Promise(resolve => setTimeout(resolve, 1000)); // Poll precisely every 1 second
  }
  throw new Error(`Workspace provisioning timed out: Pod ${resourceName} failed to achieve readiness state.`);
}

async function provisionHardenedPodAndService(resourceName: string, rid: string, branch: string) {
  const podManifest = {
    metadata: {
      name: resourceName,
      labels: { app: resourceName, type: 'workspace', 'telos.zone/tier': 'tenant-compute' },
    },
    spec: {
      nodeSelector: { 'telos.zone/role': 'ephemeral-workspaces' },
      tolerations: [{ key: 'ephemeral-only', operator: 'Exists', effect: 'NoSchedule' }],
      terminationGracePeriodSeconds: 15,
      
      // HARDENED CONFIG: Instructs Kubernetes to use your GitHub credentials
      imagePullSecrets: [
        { name: 'ghcr-cred' }
      ],
      
      volumes: [{ name: 'workspace-storage', emptyDir: { medium: 'Memory' } }], // Fast, pure in-memory execution environment
      initContainers: [
        {
          name: 'diskstemma-rehydrator',
          // Target your internal utilities image mirrored on your GitHub registry
          image: `ghcr.io/${GHCR_ORG}/telos-rehydrator:8.5.0`, 
          // Secure argument passing completely neutralizes command injection risks
          command: ['/bin/sh', '-c'],
          args: [
            'curl -fsSL -X GET "$INTERNAL_API/api/v1/internal/rehydrate/$RID/$BRANCH" -o /workspace/repo.tar && tar -xf /workspace/repo.tar -C /workspace && rm /workspace/repo.tar'
          ],
          env: [
            { name: 'INTERNAL_API', value: process.env.INTERNAL_BACKEND_URL },
            { name: 'RID', value: rid },
            { name: 'BRANCH', value: branch }
          ],
          volumeMounts: [{ name: 'workspace-storage', mountPath: '/workspace' }],
          resources: {
            requests: { cpu: '200m', memory: '256Mi' },
            limits: { cpu: '500m', memory: '512Mi' }
          }
        }
      ],
      containers: [
        {
          name: 'code-server',
          // Target your security-hardened, pre-packaged corporate VS Code image from GHCR
          image: `ghcr.io/${GHCR_ORG}/telos-code-server:4.91.1`, 
          args: ['--auth=none', '--bind-addr=0.0.0.0:8080', '--disable-telemetry', '/workspace'],
          ports: [{ containerPort: 8080, name: 'http' }],
          volumeMounts: [{ name: 'workspace-storage', mountPath: '/workspace' }],
          // Essential enterprise resource configurations
          resources: {
            requests: { cpu: '1000m', memory: '1Gi' },
            limits: { cpu: '2000m', memory: '4Gi' }
          },
          securityContext: {
            runAsNonRoot: true,
            runAsUser: 10001,
            runAsGroup: 10001,
            allowPrivilegeEscalation: false,
            readOnlyRootFilesystem: false, // Must be false for code-server extensions marketplace configurations
            capabilities: { drop: ['ALL'] }
          }
        }
      ]
    }
  };

  const serviceManifest = {
    metadata: { name: resourceName, labels: { app: resourceName } },
    spec: {
      selector: { app: resourceName },
      ports: [{ port: 8080, targetPort: 8080, protocol: 'TCP' }]
    }
  };

  // Synchronously execute resource bindings
  await k8sCoreApi.createNamespacedPod(NAMESPACE, podManifest);
  await k8sCoreApi.createNamespacedService(NAMESPACE, serviceManifest);
}
```
